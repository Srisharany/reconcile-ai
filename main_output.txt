# JSON SANITIZER
# ============================================================

def sanitize_for_json(value):
    """
    Recursively convert NaN / Infinity values to None.
    """

    if isinstance(value, dict):
        return {
            key: sanitize_for_json(val)
            for key, val in value.items()
        }

    if isinstance(value, list):
        return [
            sanitize_for_json(item)
            for item in value
        ]

    if isinstance(value, tuple):
        return [
            sanitize_for_json(item)
            for item in value
        ]

    if isinstance(value, float):
        if not math.isfinite(value):
            return None

        return value

    return value


# ============================================================
# LOAD JSON
# ============================================================

def load_json(file_path: Path):

    if not file_path.exists():
        raise HTTPException(
            status_code=404,
            detail=f"File not found: {file_path.name}",
        )

    try:
        with open(
            file_path,
            "r",
            encoding="utf-8",
        ) as file:

            data = json.load(file)

        return sanitize_for_json(data)

    except json.JSONDecodeError as error:

        raise HTTPException(
            status_code=500,
            detail=(
                f"Invalid JSON file: "
                f"{file_path.name}: {error}"
            ),
        )

# GET RECORDS

def get_records(data):

    if isinstance(data, list):
        return data

    if isinstance(data, dict):

        for key in [
            "records",
            "exceptions",
            "transactions",
            "results",
            "data",
        ]:

            value = data.get(key)

            if isinstance(value, list):
                return value

    return []


# ============================================================
# SAFE NUMBER
# ============================================================

def safe_float(value):

    if value is None:
        return None

    if isinstance(value, bool):
        return None

    try:
        number = float(value)

        if math.isfinite(number):
            return number

    except (
        TypeError,
        ValueError,
    ):
        pass

    return None


# ============================================================
# AMOUNT EXTRACTION
# ============================================================

def get_amount_at_risk(record):
    """
    Determine the financial amount associated with an exception.

    Priority:

    1. Explicit amount_at_risk
    2. amount_difference / difference
    3. Difference between source amounts
    4. Transaction/order/payment amount
    """

    # --------------------------------------------------------
    # 1. Explicit amount at risk
    # --------------------------------------------------------

    direct_risk_fields = [
        "amount_at_risk",
        "risk_amount",
        "financial_impact",
        "loss_amount",
    ]

    for field in direct_risk_fields:

        value = safe_float(record.get(field))

        if value is not None:
            return abs(value)


    # --------------------------------------------------------
    # 2. Explicit amount difference
    # --------------------------------------------------------

    difference_fields = [
        "amount_difference",
        "difference",
        "amount_diff",
        "mismatch_amount",
        "variance",
        "amount_variance",
    ]

    for field in difference_fields:

        value = safe_float(record.get(field))

        if value is not None:
            return abs(value)


    # --------------------------------------------------------
    # 3. Compare order / bank / gateway amounts
    # --------------------------------------------------------

    order_amount = None
    bank_amount = None
    gateway_amount = None


    order_fields = [
        "order_amount",
        "order_total",
        "expected_amount",
        "expected_value",
    ]

    bank_fields = [
        "bank_amount",
        "bank_transaction_amount",
        "settlement_amount",
        "settled_amount",
        "actual_amount",
    ]

    gateway_fields = [
        "gateway_amount",
        "payment_amount",
        "transaction_amount",
        "paid_amount",
    ]


    for field in order_fields:

        value = safe_float(record.get(field))

        if value is not None:
            order_amount = value
            break


    for field in bank_fields:

        value = safe_float(record.get(field))

        if value is not None:
            bank_amount = value
            break


    for field in gateway_fields:

        value = safe_float(record.get(field))

        if value is not None:
            gateway_amount = value
            break


    # --------------------------------------------------------
    # Compare available amounts
    # --------------------------------------------------------

    if order_amount is not None and bank_amount is not None:

        return abs(order_amount - bank_amount)


    if order_amount is not None and gateway_amount is not None:

        return abs(order_amount - gateway_amount)


    if bank_amount is not None and gateway_amount is not None:

        return abs(bank_amount - gateway_amount)


    # --------------------------------------------------------
    # 4. Fall back to transaction amount
    # --------------------------------------------------------

    fallback_fields = [
        "amount",
        "transaction_value",
        "value",
        "total_amount",
        "gross_amount",
    ]

    for field in fallback_fields:

        value = safe_float(record.get(field))

        if value is not None:
            return abs(value)


    # --------------------------------------------------------
    # Nothing found
    # --------------------------------------------------------

    return 0.0

@app.get("/api/summary")
def get_summary():

    report = load_json(
        FINAL_REPORT_FILE
    )


    # --------------------------------------------------------
    # Existing summary object
    # --------------------------------------------------------

    if isinstance(report, dict):

        if "total_transactions" in report:

            summary = {
                "project": "ReconcileAI",

                "total_transactions": report.get(
                    "total_transactions",
                    0,
                ),

                "reconciled": report.get(
                    "reconciled",
                    0,
                ),

                "exceptions": report.get(
                    "exceptions",
                    0,
                ),

                "exception_rate": report.get(
                    "exception_rate",
                    0,
                ),

                "amount_at_risk": report.get(
                    "amount_at_risk",
                    0,
                ),

                "currency": report.get(
                    "currency",
                    "INR",
                ),

                "severity": report.get(
                    "severity",
                    {
                        "CRITICAL": 0,
                        "HIGH": 0,
                        "MEDIUM": 0,
                        "LOW": 0,
                    },
                ),

                "ai_investigations": report.get(
                    "ai_investigations",
                    0,
                ),

                "auto_resolution": report.get(
                    "auto_resolution",
                    0,
                ),

                "pipeline": {
                    "pass_1": "Exact Matching",
                    "pass_2": "Fuzzy Matching",
                    "pass_3": "AI Investigation",
                    "pass_4": "Final Decision Engine",
                },
            }

            return sanitize_for_json(summary)


    # --------------------------------------------------------
    # Calculate from records
    # --------------------------------------------------------

    records = get_records(report)

    total_transactions = len(records)

    reconciled = 0
    exceptions_count = 0
    amount_at_risk = 0.0

    severity = {
        "CRITICAL": 0,
        "HIGH": 0,
        "MEDIUM": 0,
        "LOW": 0,
    }


    # --------------------------------------------------------
    # Process records
    # --------------------------------------------------------

    for record in records:

        if not isinstance(record, dict):
            continue


        decision = get_decision(record)


        if decision == "RECONCILED":

            reconciled += 1

            continue


        # ----------------------------------------------------
        # Exception
        # ----------------------------------------------------

        exceptions_count += 1


        risk = get_amount_at_risk(record)

        if math.isfinite(risk):

            amount_at_risk += abs(risk)


        # ----------------------------------------------------
        # Severity
        # ----------------------------------------------------

        record_severity = str(
            record.get(
                "severity",
                "",
            )
        ).strip().upper()


        if record_severity in severity:

            severity[record_severity] += 1


    # --------------------------------------------------------
    # Exception rate
    # --------------------------------------------------------

    if total_transactions > 0:

        exception_rate = round(
            (
                exceptions_count
                / total_transactions
            ) * 100,
            2,
        )

    else:

        exception_rate = 0.0


    # --------------------------------------------------------
    # AI investigation count
    # --------------------------------------------------------

    ai_investigations = 0


    if AI_INVESTIGATION_FILE.exists():

        try:

            ai_data = load_json(
                AI_INVESTIGATION_FILE
            )

            ai_records = get_records(
                ai_data
            )

            ai_investigations = len(
                ai_records
            )

        except Exception:

            ai_investigations = 0


    # --------------------------------------------------------
    # Final response
    # --------------------------------------------------------

    response = {

        "project": "ReconcileAI",

        "total_transactions":
            total_transactions,

        "reconciled":
            reconciled,

        "exceptions":
            exceptions_count,

        "exception_rate":
            exception_rate,

        "amount_at_risk":
            round(
                amount_at_risk,
                2,
            ),

        "currency":
            "INR",

        "severity":
            severity,

        "ai_investigations":
            ai_investigations,

        "auto_resolution":
            0,

        "pipeline": {

            "pass_1":
                "Exact Matching",

            "pass_2":
                "Fuzzy Matching",

            "pass_3":
                "AI Investigation",

            "pass_4":
                "Final Decision Engine",
        },
    }


    return sanitize_for_json(response)

# EXCEPTIONS
# ============================================================

@app.get("/api/exceptions")
def get_exceptions():

    report = load_json(
        FINAL_REPORT_FILE
    )

    records = get_records(report)

    exceptions = []


    for record in records:

        if not isinstance(record, dict):
            continue


        decision = get_decision(record)

        exception_type = record.get(
            "exception_type"
        )


        if (
            decision != "RECONCILED"
            or exception_type
        ):

            # ------------------------------------------------
            # Make a copy so we don't modify original data
            # ------------------------------------------------

            exception_record = dict(record)


            # ------------------------------------------------
            # Add calculated amount at risk
            # ------------------------------------------------

            calculated_risk = get_amount_at_risk(
                record
            )


            exception_record[
                "amount_at_risk"
            ] = round(
                calculated_risk,
                2,
            )


            # ------------------------------------------------
            # Ensure severity exists
            # ------------------------------------------------

            if not exception_record.get(
                "severity"
            ):

                exception_record[
                    "severity"
                ] = "MEDIUM"


            exceptions.append(
                sanitize_for_json(
                    exception_record
                )
            )


    return sanitize_for_json(
        exceptions
    )


# ============================================================
# SINGLE EXCEPTION
# ============================================================

@app.get(
    "/api/exceptions/{transaction_id}"
)
def get_exception(
    transaction_id: str
):

    report = load_json(
        FINAL_REPORT_FILE
    )

    records = get_records(report)


    for record in records:

        if not isinstance(record, dict):
            continue


        record_id = get_transaction_id(
            record
        )


        if record_id == transaction_id:

            exception_record = dict(
                record
            )


            # ------------------------------------------------
            # Calculate amount at risk
            # ------------------------------------------------

            exception_record[
                "amount_at_risk"
            ] = round(
                get_amount_at_risk(
                    record
                ),
                2,
            )


            return sanitize_for_json(
                exception_record
            )


    raise HTTPException(
        status_code=404,
        detail=(
            f"Exception not found: "
            f"{transaction_id}"
        ),
    )
